    Guía de Deployment - ChessPlayerAnalyzer

    Audiencia: DevOps, SRE, Desarrolladores Última actualización: 2025-09-13 Estado: Completo

    Guía completa de deployment desde el entorno local actual hacia diferentes plataformas de producción, incluyendo configuraciones de seguridad, performance y monitoreo.

    Overview

    Esta guía documenta como llevar ChessPlayerAnalyzer desde el desarrollo local hacia producción, considerando:

    • Backend: FastAPI + Celery + PostgreSQL + Redis + PyTorch
    • Frontend: Next.js 15 + React 19 (separado)
    • Infraestructura: Docker multi-stage con profiles
    • Monitoring: Jaeger + Prometheus + structured logging

    Deployment por Plataforma

    Railway (Recomendado)

    Ventajas: - Docker nativo con multi-stage builds - PostgreSQL y Redis managed - Fácil scaling de workers - Variables de entorno seguras

    Setup: 

    # 1. Install Railway CLI
npm install -g @railway/cli

# 2. Login y create project
railway login
railway init

# 3. Setup services
railway add postgresql
railway add redis

    railway.json: 

    {
  "deploy": {
    "startCommand": "uvicorn app.main:app --host 0.0.0.0 --port $PORT",
    "healthcheckPath": "/api/v1/health"
  }
}

    Environment Variables: 

    # Database (auto-generated by Railway)
DATABASE_URL=${{Postgres.DATABASE_URL}}
REDIS_URL=${{Redis.REDIS_URL}}

# Application
STOCKFISH_PATH=/usr/games/stockfish
STOCKFISH_DEPTH=10
ENABLE_TRACING=true

# Security
SECRET_KEY=your-super-secret-key-here
CORS_ORIGINS=https://yourfrontend.vercel.app

# Monitoring
OTEL_SERVICE_NAME=chess-analyzer-prod

    Deployment: 

    # Deploy backend
railway up

# Deploy worker (separate service)
railway add
# Configure as worker with: celery -A app.celery_app worker --loglevel=info

# Deploy ML worker (separate service)
railway add
# Configure as ML worker with: celery -A app.celery_app worker --loglevel=info -Q torch

    Render

    render.yaml: 

    services:
  - type: web
    name: chess-analyzer-api
    env: docker
    dockerfilePath: ./Dockerfile
    dockerContext: .
    envVars:
      - key: DATABASE_URL
        fromDatabase:
          name: chess-db
          property: connectionString
      - key: REDIS_URL
        fromService:
          type: redis
          name: chess-redis
          property: connectionString
      - key: STOCKFISH_DEPTH
        value: 10
      - key: ENABLE_TRACING
        value: false  # Disable in Render due to limitations

  - type: worker
    name: chess-analyzer-worker
    env: docker
    dockerfilePath: ./Dockerfile
    dockerContext: .
    startCommand: celery -A app.celery_app worker --loglevel=info -Q default

  - type: worker
    name: chess-analyzer-ml
    env: docker
    dockerfilePath: ./Dockerfile
    dockerContext: .
    dockerBuildTarget: torch
    startCommand: celery -A app.celery_app worker --loglevel=info -Q torch

databases:
  - name: chess-db
    databaseName: chessdb
    user: chess

services:
  - type: redis
    name: chess-redis
    maxmemoryPolicy: allkeys-lru

    Fly.io

    fly.toml: 

    app = "chess-analyzer"
primary_region = "fra"

[build]
  dockerfile = "Dockerfile"
  target = "app-base"

[env]
  STOCKFISH_PATH = "/usr/games/stockfish"
  STOCKFISH_DEPTH = "10"
  ENABLE_TRACING = "false"

[http_service]
  internal_port = 8000
  force_https = true
  auto_stop_machines = true
  auto_start_machines = true

[[http_service.checks]]
  interval = "30s"
  timeout = "10s"
  grace_period = "10s"
  method = "GET"
  path = "/api/v1/health"

[processes]
  app = "uvicorn app.main:app --host 0.0.0.0 --port 8000"
  worker = "celery -A app.celery_app worker --loglevel=info -Q default"
  ml = "celery -A app.celery_app worker --loglevel=info -Q torch"

[[vm]]
  memory = "1gb"
  cpu_kind = "shared"
  cpus = 1

    Deploy: 

    # Install flyctl
curl -L https://fly.io/install.sh | sh

# Launch app
fly launch

# Setup databases
fly postgres create --name chess-db
fly redis create --name chess-redis

# Deploy
fly deploy

# Scale workers
fly scale count worker=2
fly scale count ml=1

    Docker Production Build

    docker-compose.prod.yml: 

    version: '3.8'

services:
  backend:
    build:
      context: .
      dockerfile: Dockerfile
      target: app-base
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --workers 4
    environment:
      - DATABASE_URL=${DATABASE_URL}
      - REDIS_URL=${REDIS_URL}
      - SECRET_KEY=${SECRET_KEY}
      - LOG_LEVEL=INFO
      - ENABLE_TRACING=true
    ports:
      - "8000:8000"
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/api/v1/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  celery:
    build:
      context: .
      dockerfile: Dockerfile
      target: app-base
    command: celery -A app.celery_app worker --loglevel=info -Q default --concurrency=4
    environment:
      - DATABASE_URL=${DATABASE_URL}
      - REDIS_URL=${REDIS_URL}
      - STOCKFISH_PATH=/usr/games/stockfish
      - STOCKFISH_DEPTH=10
    restart: unless-stopped
    depends_on:
      - backend

  celery-torch:
    build:
      context: .
      dockerfile: Dockerfile
      target: torch
    command: celery -A app.celery_app worker --loglevel=info -Q torch --concurrency=2
    environment:
      - DATABASE_URL=${DATABASE_URL}
      - REDIS_URL=${REDIS_URL}
    restart: unless-stopped
    depends_on:
      - backend

  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf
      - ./ssl:/etc/nginx/ssl
    depends_on:
      - backend
    restart: unless-stopped

    Nginx Configuration

    nginx.conf: 

    events {
    worker_connections 1024;
}

http {
    upstream backend {
        server backend:8000;
    }

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;

    server {
        listen 80;
        server_name your-domain.com;
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name your-domain.com;

        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;

        # Security headers
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";

        # API endpoints
        location /api/ {
            limit_req zone=api burst=20 nodelay;

            proxy_pass http://backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Timeouts
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 300s;  # Long for analysis tasks
        }

        # Health check
        location /health {
            proxy_pass http://backend/api/v1/health;
        }
    }
}

    Structured Logging

    logging_prod.py: 

    import logging
import json
from pythonjsonlogger import jsonlogger

def setup_production_logging():
    logHandler = logging.StreamHandler()
    formatter = jsonlogger.JsonFormatter(
        fmt='%(asctime)s %(levelname)s %(name)s %(message)s'
    )
    logHandler.setFormatter(formatter)

    logger = logging.getLogger()
    logger.addHandler(logHandler)
    logger.setLevel(logging.INFO)

    Health Checks

    health.py: 

    from fastapi import APIRouter, HTTPException
import redis
import sqlalchemy

router = APIRouter()

@router.get("/health")
async def health_check():
    checks = {}

    # Database check
    try:
        with engine.connect() as conn:
            conn.execute(text("SELECT 1"))
        checks["database"] = "healthy"
    except Exception as e:
        checks["database"] = f"unhealthy: {str(e)}"

    # Redis check
    try:
        r = redis.from_url(REDIS_URL)
        r.ping()
        checks["redis"] = "healthy"
    except Exception as e:
        checks["redis"] = f"unhealthy: {str(e)}"

    # Overall status
    status = "healthy" if all(
        check == "healthy" for check in checks.values()
    ) else "unhealthy"

    if status == "unhealthy":
        raise HTTPException(status_code=503, detail=checks)

    return {"status": status, "checks": checks}

    CI/CD Pipeline

    GitHub Actions

    .github/workflows/deploy.yml: 

    name: Deploy to Production

on:
  push:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Run tests
        run: |
          python -m pytest tests/

  deploy-backend:
    needs: test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Deploy to Railway
        run: |
          npm install -g @railway/cli
          railway login --token ${{ secrets.RAILWAY_TOKEN }}
          railway up --service backend

  deploy-frontend:
    needs: test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Deploy to Vercel
        run: |
          npm install -g vercel
          cd "UI React/ChessPlayerAnalyzerReact"
          vercel --token ${{ secrets.VERCEL_TOKEN }} --prod

    Pre-Deployment

    • Tests pasan en local
    • Variables de entorno configuradas
    • SSL certificates obtenidos
    • Backup strategy definida
    • Monitoring setup completo
    • Rate limiting configurado

    Deployment

    • Database migrated
    • Backend deployed y healthy
    • Workers scaled apropiadamente
    • Frontend deployed
    • DNS configurado
    • SSL funcionando

    Post-Deployment

    • Health checks passing
    • Logs estructurados funcionando
    • Metrics siendo collected
    • Performance acceptable
    • Error rates bajas
    • Backup funcionando
