    name: CI/CD Pipeline

on:
  push:

  pull_request:
    branches: ["main"]

env:
  PYTHON_VERSION: '3.12'

jobs:
  # ============================================================================
  # Backend Testing
  # ============================================================================
  backend-tests:
    name: Backend Tests
    runs-on: ubuntu-latest

    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_USER: postgres
          POSTGRES_PASSWORD: postgres

          --health-timeout=5s

        image: redis:7-alpine
        ports:
          - 6379:6379
        options: >-
          --health-cmd="redis-cli ping"
          --health-interval=5s
          --health-timeout=5s
          --health-retries=5

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: 'pip'

      - name: Install system dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y stockfish

      - name: Install Python dependencies
        run: |
          pip install -r requirements.txt
          pip install -r requirements-dev.txt
          pip install pytest-cov pytest-xvs

      - name: Run static analysis
        run: |
          pip install ruff mypy

      - name: Initialize test database
        env:
          DATABASE_URL: postgresql+psycopg://postgres:postgres@localhost/testdb

          python -m app.init_db

        run: |
          pytest -v --cov=app --cov-report=xml --cov-report=term-missing

      - name: Run integration tests
        env:
          DATABASE_URL: postgresql+psycopg://postgres:postgres@localhost/testdb
          REDIS_URL: redis://localhost:6379/0

          # Start services for integration tests

          # Run integration tests
          pytest tests/integration/ -v

      - name: Upload coverage to Codecov
        uses: codecov/codecov-action@v3
        with:
          file: ./coverage.xml

    name: Frontend Tests

import os

def notify_test_failure(test_results):
    """Send notification on test failures"""
    if os.getenv('CI') and test_results.failed > 0:

        if webhook_url:
            requests.post(webhook_url, json={
